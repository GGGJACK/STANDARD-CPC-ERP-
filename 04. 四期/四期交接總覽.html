5. 隔离复测方法

以下命令仅示例，路径请换成四期实际位置。第一个窗口启动测试服务，第二个窗口运行验收；测试会写数据，所以必须使用副本。

REM 准备：在 系统代码 目录复制 data.db 为 cpc-test.db，并新建 cpc-test-uploads
set PORT=3100
set DB_PATH=四期目录\系统代码\cpc-test.db
set CPC_UPLOAD_DIR=四期目录\系统代码\cpc-test-uploads
set CPC_DEMO=1
node server.mjs

REM 另一个窗口
set CPC_TEST=1
set TEST_BASE=http://127.0.0.1:3100
set DB_PATH=四期目录\系统代码\cpc-test.db
set TEST_OUTPUT=四期目录\系统代码\cpc-test-results.json
node test.mjs
